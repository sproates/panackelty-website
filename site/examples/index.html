Ordered conditional chains

Each condition is checked in order. Only the first matching body runs.


pure classify(value: Nat): Str {
  if value == 0 {
    "zero"
  } else if value == 1 {
    "one"
  } else if value == 2 {
    "two"
  } else {
    "many"
  }
}

main(): Void {
  print(classify(0))
  print(classify(2))
  print(classify(7))
}

Output


zero
two
many

Give a business rule a name

Positive expresses a constraint on an integer. Code receiving this type can state its expectation directly.


type Positive = Int where value > 0

pure triple(n: Positive): Int {
  n * 3
}

main(): Void {
  answer: Positive = 14
  print(triple(answer))
}

Output


42

Handle a failure explicitly

A Result carries either a value or an error. Exhaustive matching makes both paths visible.


pure safe_divide(numerator: Nat, denominator: Nat): Result[Nat,Str] {
  if denominator == 0 {
    Error("division by zero")
  } else {
    Ok(quotient(numerator, denominator))
  }
}

pure describe(result: Result[Nat,Str]): Str {
  match result {
    Ok(value) => "value ${value}",
    Error(message) => message
  }
}

main(): Void {
  print(describe(safe_divide(84, 2)))
  print(describe(safe_divide(1, 0)))
}

Output


value 42
division by zero

Keep the original collection

Appending returns a new array. Updating a persistent map also returns a new value; assign it when you want to keep the change.


main(): Void {
  original: [Nat] = [1, 2]
  extended: [Nat] = original.append(3)

  print(original)
  print(extended)

  mut scores: Map[Str,Nat] = map()
  scores = scores.put("Ada", 42)
  print(scores.get("Ada"))
}

Output


[1, 2]
[1, 2, 3]
42

Describe a task with a record

Give related values named fields, then keep presentation in a pure function.


record Task {
  title: Str,
  done: Bool
}

pure describe_task(task: Task): Str {
  title = task.title

  if task.done {
    "Done: ${title}"
  } else {
    "To do: ${title}"
  }
}

main(): Void {
  task = Task("Try Panackelty", false)
  print(describe_task(task))
}

Output


To do: Try Panackelty

Sort and select filenames

Pass a named pure comparison function to sort_by, then use a string predicate to select matching names. This processes names; it does not read the filesystem.


import stdlib/prelude

pure name_before(left: Str, right: Str): Bool {
  left < right
}

main(): Void {
  files = ["z.txt", "b.panack", "a.panack"].sort_by(@name_before)

  for file in files {
    if file.ends_with(".panack") {
      print(file)
    }
  }
}

Output


a.panack
b.panack

Build a result one step at a time

Local mutation can be useful inside a pure calculation. Each iteration advances the two values in the sequence.


pure fibonacci(n: Nat): Nat {
  mut previous: Nat = 0
  mut current: Nat = 1

  for ignored in 0..n {
    next: Nat = previous + current
    previous = current
    current = next
  }

  previous
}

main(): Void {
  print(fibonacci(10))
}

Output


55

Native alpha.13

Serve your first
HTTP response.

This complete released example answers one request, then exits. Its handler returns a typed response containing UTF-8 text. The server admits one client, bounds request and response sizes, and has explicit deadlines.

Install native alpha.13 first. Obtain the matching example and its local HTTP package in a new directory; Git is needed for this step:

git clone --branch v0.1.0-alpha.13 --depth 1 https://github.com/sproates/panackelty.git panackelty-alpha13-examples
cd panackelty-alpha13-examples
panack run --project examples/network/http_server/panack.json

In a second terminal, request the response. If the connection is refused before the program has bound its port, wait briefly and try again:

curl --fail http://127.0.0.1:8080/

The response is hello from Panackelty HTTP server. The server then exits successfully. Restart it before another request. This is a finite cleartext example; it does not provide server TLS.

Read the complete server program
// Start this program, then request http://127.0.0.1:8080/ once. The finite
// server closes after its first client has finished.
import http/server as http
import stdlib/text as text

async respond(request: http.Request): Result[http.Response,http.HttpError] {
  Ok(http.Response(200,
    [http.Header("Content-Type", "text/plain; charset=utf-8")],
    text.encode_utf8("hello from Panackelty HTTP server\n")))
}

pure expect(value: Bool): Unit {
  checked = [0][if value { 0 } else { 1 }]
  ()
}

async main(): Unit {
  result = await http.serve("127.0.0.1", 8080, @respond,
    http.ServerLimits(1, 1, 4096, 4096, 5000, 10000, 1000))
  match result {
    Error(problem) => expect(false),
    Ok(reports) => {
      expect(reports.len() == 1)
      match reports[0] { Ok(done) => (), Error(problem) => expect(false) }
    }
  }
  ()
}

Native alpha.13

Use a local
package.

The HTTP example above is also a complete local-package example. Its application manifest gives the HTTP library the alias http; the source imports http/server through that library’s public exports.

Here is examples/network/http_server/panack.json, formatted for reading:

{
  "schema": 1,
  "package": "examples.http_server",
  "version": "0.1.0",
  "language": "modules-v1",
  "toolchain": "0.1.0-alpha.13",
  "source": "src",
  "entry": "main.panack",
  "exports": {},
  "dependencies": {
    "http": {
      "path": "../../../packages/http"
    }
  }
}

The dependency path is relative to the manifest. It reaches packages/http inside the same checkout. The application’s source directory is src, its entry is main.panack, and its toolchain must match alpha.13 exactly. Keep this directory structure together when moving the example.

From the checkout root, check it without starting the server:

panack check --project examples/network/http_server/panack.json

The HTTP library exports client, server, service and types. Other files remain implementation details. This example uses a local dependency; no registry or remote package restoration is involved.

Native alpha.13

Preview a folder
in your browser.

The native installation includes panack-http-server. Create a small site and serve it without a source checkout:

mkdir preview-demo
printf '%s\n' '<h1>Hello from Panackelty</h1>' > preview-demo/index.html
panack-http-server --root ./preview-demo --port 4173 --no-cache --noindex

Wait for Serving http://127.0.0.1:4173/, then open that address in your browser. Edit index.html and refresh to see your change. Press Ctrl-C to stop; active requests have up to one second to drain.

--root chooses the served folder. --no-cache asks the browser not to store responses, and --noindex asks search engines not to index them. The default bind address is loopback, so other devices cannot connect. Use another free port if 4173 is already occupied.

The tool serves existing files and directory indexes with confined reads. It does not list directories, rebuild generated sites, watch files or provide TLS. Complete responses must fit within its 1 MiB limit.
